async function saveItem() { const name = document.getElementById('itemName').value.trim(); if(!name) { alert('Please enter an item name.'); return; } const editId = document.getElementById('editItemId').value; const itemData = { name: name, category: document.getElementById('itemCategory').value, store: document.getElementById('itemStore').value.trim() || 'General', price: parseFloat(document.getElementById('itemPrice').value) || 0, status: document.getElementById('itemStatus').value, inBasket: false, image: document.getElementById('itemImage').value.trim(), notes: document.getElementById('itemNotes').value.trim() }; try { await window.firebaseAddOrUpdateItem(itemData, editId); closeModal(); } catch (error) { alert('Error: ' + error.message); console.error(error); } }
